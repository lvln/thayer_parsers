rulelist        =  1*( rule / (*wsp c-nl) )

rule            =  rulename defined-as elements c-nl
                ; continues if next line starts
                ;  with white space

rulename        =  ALPHA *(ALPHA / DIGIT / "-")

defined-as      =  *c-wsp ("=" / "=/") *c-wsp
                ; basic rules definition and
                ;  incremental alternatives

elements        =  alternation *wsp

c-wsp           =  WSP / (c-nl WSP)

c-nl            =  comment / CRLF
                ; comment or newline

comment         =  ";" *(WSP / VCHAR) CRLF

alternation     =  concatenation
                   *(*c-wsp "/" *c-wsp concatenation)

concatenation   =  repetition *(1*c-wsp repetition)

repetition      =  [repeat] element

repeat          =  1*DIGIT / (*DIGIT "*" *DIGIT)

element         =  rulename / group / option /
                   char-val / num-val / prose-val

group           =  "(" *c-wsp alternation *c-wsp ")"

option          =  "[" *c-wsp alternation *c-wsp "]"

char-val        =  DQUOTE *(%x20-21 / %x23-7E) DQUOTE
                ; quoted string of SP and VCHAR
                ;  without DQUOTE

num-val         =  "%" (bin-val / dec-val / hex-val / int-val)

bin-val         =  "b" 1*BIT
                   [ 1*("." 1*BIT) / ("-" 1*BIT) ]
                ; series of concatenated bit values
                ;  or single ONEOF range

dec-val         =  "d" 1*DIGIT
                   [ 1*("." 1*DIGIT) / ("-" 1*DIGIT) ]

hex-val         =  "x" 1*HEXDIG
                   [ 1*("." 1*HEXDIG) / ("-" 1*HEXDIG) ]

prose-val       =  "<" *(%x20-3D / %x3F-7E) ">"
                ; bracketed string of SP and VCHAR
                ;  without angles
                ; prose description, to be used as
                ;  last resort

int-val         =  uint8 / int8 / uint16-le / int16-le / uint16-be / int16-be / uint32-le
                   / int32-le / uint32-be / int32-be / uint64-le / int64-le / uint64-be / int64-be

uint8           =  "u8_" u8_value
                   [ 1*("." u8-value) / ("-" u8-value) ]

int8            =  "i8_" i8-value
                   [ 1*("." i8-value) / ("-" i8-value) ]

uint16-le       =  "u16_le_" u16-value
                   [ 1*("." u16-value) / ("-" u16-value) ]

int16-le        =  "i16_le_" i16-value
                   [ 1*("." i16-value) / ("-" i16-value) ]

uint16-be       =  "u16_be_" u16-value
                   [ 1*("." u16-value) / ("-" u16-value) ]

int16-be        =  "i16_be_" i16-value
                   [ 1*("." i16-value) / ("-" i16-value) ]

uint32-le       =  "u32_le_" u32-value
                   [ 1*("." u32-value) / ("-" u32-value) ]

int32-le        =  "i32_le_" i32-value
                   [ 1*("." i32-value) / ("-" i32-value) ]

uint32-be       =  "u32_be_" u32-value
                   [ 1*("." u32-value) / ("-" u32-value) ]

int32-be        =  "i32_be_" i32-value
                   [ 1*("." i32-value) / ("-" i32-value) ]

uint64-le       =  "u64_le_" u64-value
                   [ 1*("." u64-value) / ("-" u64-value) ]

int64-le        =  "i64_le_" i64-value
                   [ 1*("." i64-value) / ("-" i64-value) ]

uint64-be       =  "u64_be_" u64-value
                   [ 1*("." u64-value) / ("-" u64-value) ]

int64-be        =  "i64_be_" i64-value
                   [ 1*("." i64-value) / ("-" i64-value) ]

ALPHA           =  %x41-5A / %x61-7A
                ; A-Z / a-z

BIT             =  "0" / "1"

CHAR            =  %x01-7F
                ; any 7-bit US-ASCII character,
                ;  excluding NUL

CR              =  %x0D
                ; carriage return

CRLF            =  CR LF
                ; Internet standard newline

CTL             =  %x00-1F / %x7F
                ; controls

DIGIT           =  %x30-39
                ; 0-9

DQUOTE          =  %x22
                ; " (Double Quote)

HEXDIG          =  DIGIT / "A" / "B" / "C" / "D" / "E" / "F"

HTAB            =  %x09
                ; horizontal tab

LF              =  %x0A
                ; linefeed

LWSP            =  *(WSP / CRLF WSP)
                ; Use of this linear-white-space rule
                ;  permits lines containing only white
                ;  space that are no longer legal in
                ;  mail headers and have caused
                ;  interoperability problems in other
                ;  contexts.
                ; Do not use when defining mail
                ;  headers and use with caution in
                ;  other contexts.

OCTET           =  %x00-FF
                ; 8 bits of data

SP              =  %x20

VCHAR           =  %x21-7E
                ; visible (printing) characters

WSP             =  SP / HTAB
                ; white space

ONENINE         =  %d31-39

u8-value        =  DIGIT
                   / ONENINE DIGIT
                   / "1" 2DIGIT
                   / "2" (%x30-34 DIGIT
                   / "5" %x30-35)

i8-value        =   [-] (DIGIT
                   / ONENINE DIGIT
                   / "1" (%x30-31 DIGIT
                   / "2" %x30-37))
                   / "-128"

u16-value       =  DIGIT
                   / ONENINE 1*3DIGIT
                   / "6" (%x30-34 3DIGIT
                   / "5" (%x30-34 2DIGIT
                   / "5" (%x30-32 DIGIT
                   / "3" %x30-35)))

i16-value       =  [-] (DIGIT
                   / ONENINE 1*3DIGIT
                   / "3" (%30-31 3DIGIT
                   / "2" (%30-36 2DIGIT
                   / "7" (%30-35 DIGIT
                   / "6" %x30-37))))
                   / "-32768"

u32-value       =  DIGIT
                   / ONENINE 1*8DIGIT
                   / "4" (%x30-31 8DIGIT
                   / "2" (%x30-38 7DIGIT
                   / "9" (%x30-33 6DIGIT
                   / "4" (%x30-38 5DIGIT
                   / "9" (%x30-35 4DIGIT
                   / "6" (%x30-36 3DIGIT
                   / "7" (%x30-31 2DIGIT
                   / "2" (%x30-38 DIGIT
                   / "9" %x30-35)))))))))

i32-value       =  [-] (DIGIT
                   / ONENINE 1*8DIGIT
                   / "2" (%x30 8DIGIT
                   / "1" (%x30-33 7DIGIT
                   / "4" (%x30-36 6DIGIT
                   / "7" (%x30-33 5DIGIT
                   / "4" (%x30-37 4DIGIT
                   / "8" (%x30-32 3DIGIT
                   / "3" (%x30-35 2DIGIT
                   / "6" (%x30-33 DIGIT
                   / "4" %x30-37)))))))))
                   / "-2147483648"

u64-value       =  DIGIT
                   / ONENINE 1*18DIGIT
                   / "1" (%x30-37 18DIGIT
                   / "8" (%x30-33 17DIGIT
                   / "4" (%x30-33 16DIGIT
                   / "4" (%x30-35 15DIGIT
                   / "6" (%x30-36 14DIGIT
                   / "7" (%x30-33 13DIGIT
                   / "4" (%x30-33 12DIGIT
                   / "40" (%x30-36 10DIGIT
                   / "7" (%x30-32 9DIGIT
                   / "3" (%x30-36 8DIGIT
                   / "70" (%x30-38 6DIGIT
                   / "9" (%x30-34 5DIGIT
                   / "5" (%x30-34 4DIGIT
                   / "5" (%x30 3DIGIT
                   / "1" (%x30-35 2DIGIT
                   / "6" (%x30 DIGIT
                   / "1" %x30-35)))))))))))))))))))

i64-value       =  [-] (DIGIT
                   / ONENINE 1*17DIGIT
                   / "9" (%x30-31 17DIGIT
                   / "2" (%x30-31 16DIGIT
                   / "2" (%x30-32 15DIGIT
                   / "3" (%x30-32 14DIGIT
                   / "3" (%x30-36 13DIGIT
                   / "7" (%x30-31 12DIGIT
                   / "20" (%x30-32 10DIGIT
                   / "3" (%x30-35 9DIGIT
                   / "6" (%x30-37 8DIGIT
                   / "8" (%x30-34 7DIGIT
                   / "5" (%x30-33 6DIGIT
                   / "4" (%x30-36 5DIGIT
                   / "7" (%x30-36 4DIGIT
                   / "7" (%x30-34 3DIGIT
                   / "5" (%x30-37 2DIGIT
                   / "80" %x30-37))))))))))))))))
                   / "-9223372036854775808"
